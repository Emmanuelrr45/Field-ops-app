// Add this function inside your index.html script to check agent activity async function checkAgentMonthlyActivity(agentName) { if (!agentName) return; const currentMonthKey = new Date().toISOString().slice(0, 7); // e.g., "2026-11" try { const { data, error } = await supabaseAgent .from('visits') .select('created_at') .ilike('agent_name', agentName.trim()); if (error) throw error; // Filter visits made in the current month by this agent const currentMonthVisits = (data || []).filter(v => v.created_at && v.created_at.slice(0, 7) === currentMonthKey); const warningContainer = document.getElementById('agentActivityWarning'); if (!warningContainer) return; if (currentMonthVisits.length === 0) { warningContainer.innerHTML = `
⚠️ Reminder: You have not logged any field visits for this month yet. Time to head out for field operations and follow-ups!
`; warningContainer.classList.remove('hidden'); } else { warningContainer.classList.add('hidden'); } } catch (err) { console.error('Error checking agent activity:', err); } }
